בנה אפליקציית אינטרנט בעברית מלאה, מימין לשמאל, בשם "תיעוד אירועי אלימות שבועי".
האפליקציה מיועדת למנתחות התנהגות שממלאות אותה בנייד בזמן אמת במהלך יום הלימודים. הנייד הוא המסך העיקרי.
מצורף למטה קוד HTML של גרסה קיימת. השתמש בו כדוגמה מדויקת לעיצוב, למבנה ולהתנהגות, אבל החלף את מנגנון השמירה שלו במסד נתונים בענן ובכניסה עם מייל, כמתואר כאן.

כניסה ושמירת נתונים לפי מייל
- כל משתמשת נרשמת ונכנסת עם כתובת המייל שלה (קישור כניסה למייל או מייל וסיסמה).
- כל הנתונים שהיא רושמת נשמרים במסד נתונים בענן תחת החשבון שלה ומסונכרנים בין כל המכשירים שבהם היא מחוברת.
- כל משתמשת רואה רק את הנתונים שלה. משתמשת אחרת לא רואה אותם.
- כפתור "שליחת הסיכום למייל שלי" ששולח את הסיכום השבועי (טקסט מסודר וקובץ CSV מצורף) לכתובת המייל של המשתמשת המחוברת.
- שמירה אוטומטית אחרי כל הקשה, עם חיווי "נשמר". בלי קליטה — לשמור במכשיר ולסנכרן כשהחיבור חוזר.

תיקים
- משתמשת יכולה לפתוח כמה תיקים (תלמיד/ה או מסגרת). לכל תיק אותו מבנה שבועי.
- בשם התיק לרשום כינוי או ראשי תיבות בלבד, לא שם מלא של תלמיד.

מסך ראשי
- כותרת: "תיעוד אירועי אלימות שבועי". שורת משנה: "דף איסוף נתונים · התנהגות: אלימות".
- בחירת תיק, מקרא צבעוני של שלוש ההתנהגויות, ושורת מעבר בין שבועות (קודם / הבא) עם טווח התאריכים.
- שורת שישה ימים עם תאריך ומספר האירועים בכל יום. בשבוע הנוכחי האפליקציה נפתחת על היום של היום.
- רשימת עשר המשבצות. בכל משבצת: שעה ושם, ושלושה כפתורי הקשה גדולים (מכות / בעיטות / דחיפות) שמציגים את המספר. המשבצת של השעה הנוכחית מודגשת.
- כפתור "הערה" בכל משבצת שפותח תיבת טקסט.
- סרגל קבוע בתחתית המסך: "ביטול פעולה אחרונה", "מצב הפחתה" (כל הקשה מפחיתה אחד, לא מתחת לאפס) וסיכום היום.

סיכום שבועי
- ארבע קוביות: סה״כ, מכות, בעיטות, דחיפות.
- טבלה: משבצות מול ימים, עם סה״כ לשורה ולעמודה והצללה לפי כמות האירועים.
- "העתקת הסיכום" — טקסט מסודר להדבקה בוואטסאפ.
- "הורדת קובץ לגיליון" — CSV עם העמודות: תאריך, יום, שעה, משבצת, מכות בידיים, בעיטות, דחיפות, סה״כ, הערה (עם BOM כדי שאקסל יציג עברית).
- "שליחת הסיכום למייל שלי".

    }finally{saving=false; if(dirty) flush()}
  }

  function subscribe(){
    if(unsub){unsub();unsub=null}
    const local=loadLocal();
    state=local||{label:'',cells:{}};
    render();
    if(!db) return;
    unsub=db.doc('weeks/'+iso(week)).onSnapshot(snap=>{
      if(snap.metadata.hasPendingWrites) return;
      if(dirty||saving||Date.now()-lastLocalEdit<1500) return;
      if(snap.exists){
        const d=snap.data();
        state={label:d.label||'',cells:JSON.parse(JSON.stringify(d.cells||{}))};
        saveLocal();
      }else if(local && Object.keys(local.cells||{}).length){
        dirty=true; flush();
      }
      setStatus('ok','נשמר בענן · מסונכרן בין מכשירים');
      render();
    },err=>{
      mode='local'; setStatus('warn','החיבור לענן נותק — הנתונים נשמרים במכשיר הזה');
    });
  }

  /* ---------- תצוגה ---------- */
  function renderDays(){
    const el=$('days'); el.innerHTML='';
    DAYS.forEach((name,i)=>{
      const d=addDays(week,i);
      let t=0; SLOTS.forEach(s=>t+=sum(cell(i,s.id)));
      const b=document.createElement('button');
      b.className='day'; b.setAttribute('aria-pressed',i===dayIdx?'true':'false');
      b.innerHTML='<b></b><small></small><span class="n"></span>';
      b.children[0].textContent=name; b.children[1].textContent=fmt(d);
      b.children[2].textContent=t?t+' אירועים':'—';
      b.onclick=()=>{dayIdx=i;openSlot=null;render()};
      el.appendChild(b);
    });
  }

  function isNow(s){
    const d=addDays(week,dayIdx); if(iso(d)!==iso(new Date())) return false;
    const n=new Date(); const m=n.getHours()*60+n.getMinutes();
    return m>=mins(s.s)&&m<mins(s.e);
  }

  function renderSlots(){
    const el=$('slots'); el.innerHTML='';
    SLOTS.forEach(s=>{
      const c=cell(dayIdx,s.id);
      const row=document.createElement('div');
      row.className='slot'+(s.br?' break':'')+(isNow(s)?' now':'');
      const when=document.createElement('div'); when.className='when';
      when.innerHTML='<span></span><b></b><button type="button"></button>';
      when.children[0].textContent=s.t; when.children[1].textContent=s.n;
      when.children[2].textContent=openSlot===s.id?'סגירה':(c.n?'הערה ✎':'הערה');
      when.children[2].onclick=()=>{openSlot=openSlot===s.id?null:s.id;renderSlots()};
      row.appendChild(when);
      const ctrs=document.createElement('div'); ctrs.className='counters';
      B.forEach(b=>{
        const v=c[b.k]||0;
        const btn=document.createElement('button');
        btn.type='button';
        btn.className='ctr '+b.k+(v?'':' zero')+(minus?' minus':'');
        btn.setAttribute('aria-label',b.n+' ב'+s.n+': '+v+(minus?' — הקשה תפחית':' — הקשה תוסיף'));
        btn.innerHTML='<span></span><strong></strong>';
        btn.children[0].textContent=b.short; btn.children[1].textContent=v;
        btn.onclick=()=>bump(dayIdx,s.id,b.k,minus?-1:1);
        ctrs.appendChild(btn);
      });
      row.appendChild(ctrs);
      if(openSlot===s.id){
        const det=document.createElement('div'); det.className='detail';
        const ta=document.createElement('textarea');
        ta.id='note-'+s.id; ta.placeholder='הערה: מה קרה לפני, כלפי מי, מה קרה אחרי…';
        ta.value=c.n||'';
        ta.oninput=()=>{const cc=Object.assign({},cell(dayIdx,s.id));cc.n=ta.value;setCell(dayIdx,s.id,cc);scheduleSave()};
        det.appendChild(ta); row.appendChild(det);
        setTimeout(()=>{ta.focus();const L=ta.value.length;try{ta.setSelectionRange(L,L)}catch(e){}},0);
      }else if(c.n){
        const p=document.createElement('p'); p.className='note-preview'; p.textContent='הערה: '+c.n; row.appendChild(p);
      }
      el.appendChild(row);
    });
    let t={h:0,k:0,p:0}; SLOTS.forEach(s=>{const c=cell(dayIdx,s.id);B.forEach(b=>t[b.k]+=c[b.k]||0)});
    $('dayTot').textContent='יום '+DAYS[dayIdx]+': '+(t.h+t.k+t.p)+' (מכות '+t.h+' · בעיטות '+t.k+' · דחיפות '+t.p+')';
  }

  function renderSummary(){
    const T={h:0,k:0,p:0};
    let maxCell=0;
    SLOTS.forEach(s=>DAYS.forEach((_,i)=>{const c=cell(i,s.id);B.forEach(b=>T[b.k]+=c[b.k]||0);maxCell=Math.max(maxCell,sum(c))}));
    $('totals').innerHTML='';
    [[ 'tot-all','סה״כ',T.h+T.k+T.p ],['c-h','מכות',T.h],['c-k','בעיטות',T.k],['c-p','דחיפות',T.p]].forEach(([cls,l,v])=>{
      const d=document.createElement('div'); d.className='tot-box '+cls; d.innerHTML='<b></b><span></span>';
      d.children[0].textContent=v; d.children[1].textContent=l; $('totals').appendChild(d);
    });
    const tbl=$('grid');
    let h='<thead><tr><th>משבצת</th>'+DAYS.map((d,i)=>'<th>'+d+'<br><small>'+fmt(addDays(week,i))+'</small></th>').join('')+'<th>סה״כ</th></tr></thead><tbody>';
    const colT=[0,0,0,0,0,0];
    SLOTS.forEach(s=>{
      let rowT=0;
      h+='<tr><td class="rowh">'+s.n+'<small>'+s.t+'</small></td>';
      DAYS.forEach((_,i)=>{
        const c=cell(i,s.id), v=sum(c); rowT+=v; colT[i]+=v;
        const a=maxCell?(v/maxCell)*0.55:0;
        const title=v?('מכות '+(c.h||0)+' · בעיטות '+(c.k||0)+' · דחיפות '+(c.p||0)):'';
        h+='<td class="cell" title="'+title+'" style="background:rgba(var(--heat),'+a.toFixed(2)+')">'+(v||'')+'</td>';
      });
      h+='<td><b>'+rowT+'</b></td></tr>';
    });
    h+='</tbody><tfoot><tr><td class="rowh">סה״כ</td>'+colT.map(v=>'<td>'+v+'</td>').join('')+'<td>'+colT.reduce((a,b)=>a+b,0)+'</td></tr></tfoot>';
    tbl.innerHTML=h;
  }

  function render(){
    const end=addDays(week,5);
    $('weekTitle').textContent='שבוע '+fmt(week)+' – '+fmt(end)+'.'+end.getFullYear();
    const who=$('who'); if(document.activeElement!==who) who.value=state.label||'';
    renderDays(); renderSlots(); renderSummary();
    $('undoBtn').disabled=!undo.length;
  }

  /* ---------- פעולות ---------- */
  function bump(d,s,k,delta){
    const c=Object.assign({h:0,k:0,p:0,n:''},cell(d,s));
    const nv=Math.max(0,(c[k]||0)+delta);
    if(nv===(c[k]||0)) return;
    undo.push({week:iso(week),d,s,k,prev:c[k]||0});
    if(undo.length>50) undo.shift();
    c[k]=nv; setCell(d,s,c); scheduleSave(); render();
    if(navigator.vibrate){try{navigator.vibrate(15)}catch(e){}}
  }
  $('undoBtn').onclick=()=>{
    const u=undo.pop(); if(!u) return;
    if(u.week!==iso(week)){render();return}
    const c=Object.assign({h:0,k:0,p:0,n:''},cell(u.d,u.s)); c[u.k]=u.prev; setCell(u.d,u.s,c); scheduleSave(); render();
  };
  $('minusBtn').onclick=()=>{minus=!minus;$('minusBtn').setAttribute('aria-pressed',minus);renderSlots()};
  $('prevW').onclick=()=>{flush();week=addDays(week,-7);dayIdx=0;undo=[];openSlot=null;subscribe()};
  $('nextW').onclick=()=>{flush();week=addDays(week,7);dayIdx=0;undo=[];openSlot=null;subscribe()};
  $('who').oninput=e=>{state.label=e.target.value;scheduleSave()};

  function summaryText(){
    let t='תיעוד אירועי אלימות שבועי\n'+(state.label?state.label+'\n':'')+$('weekTitle').textContent+'\n\n';
    DAYS.forEach((dn,i)=>{
      const lines=[];
      SLOTS.forEach(s=>{const c=cell(i,s.id);if(sum(c)||c.n){lines.push('  '+s.t+' '+s.n+': מכות '+(c.h||0)+', בעיטות '+(c.k||0)+', דחיפות '+(c.p||0)+(c.n?' — '+c.n:''))}});
      t+=dn+' '+fmt(addDays(week,i))+(lines.length?'\n'+lines.join('\n'):': אין אירועים')+'\n';
    });
    const T={h:0,k:0,p:0}; SLOTS.forEach(s=>DAYS.forEach((_,i)=>{const c=cell(i,s.id);B.forEach(b=>T[b.k]+=c[b.k]||0)}));
    t+='\nסה״כ לשבוע: '+(T.h+T.k+T.p)+' (מכות '+T.h+', בעיטות '+T.k+', דחיפות '+T.p+')';
    return t;
  }
  $('copyBtn').onclick=async()=>{
    const txt=summaryText();
    try{await navigator.clipboard.writeText(txt);$('msg').textContent='הסיכום הועתק. אפשר להדביק בוואטסאפ או במסמך.'}
    catch(e){
      const ta=document.createElement('textarea');ta.value=txt;ta.style.width='100%';ta.rows=8;$('msg').textContent='ההעתקה האוטומטית לא זמינה כאן — סמני את הטקסט והעתיקי:';$('msg').after(ta);ta.select();
    }
  };
  function csv(){
    const q=v=>'"'+String(v).replace(/"/g,'""')+'"';
    const rows=[['תאריך','יום','שעה','משבצת','מכות בידיים','בעיטות','דחיפות','סה״כ','הערה']];
    DAYS.forEach((dn,i)=>SLOTS.forEach(s=>{const c=cell(i,s.id);rows.push([iso(addDays(week,i)),dn,s.t,s.n,c.h||0,c.k||0,c.p||0,sum(c),c.n||''])}));
    return '﻿'+rows.map(r=>r.map(q).join(',')).join('\r\n');
  }

  /* ---------- אתחול ---------- */
  subscribe();
  setStatus('','שמירה במכשיר — בודק חיבור לענן…');
  (async()=>{
    try{
      for(let i=0;i<30&&!(window.claude&&window.claude.use);i++) await new Promise(r=>setTimeout(r,100));
      if(window.claude&&window.claude.use){
        db=await window.claude.use('db');
        const dl=await window.claude.use('downloads');
        if(dl){
          $('csvBtn').hidden=false;
          $('csvBtn').onclick=async()=>{
            try{await dl.save({filename:'איסוף-נתונים-'+iso(week)+'.csv',data:csv()});$('msg').textContent='הקובץ נשמר.'}
            catch(e){$('msg').textContent='הקובץ לא נשמר.'}
          };
        }
      }
    }catch(e){db=null}
    if(db){mode='cloud';setStatus('warn','מתחבר לענן…');subscribe()}
    else setStatus('warn','הנתונים נשמרים במכשיר הזה בלבד (אין חיבור לענן בתצוגה זו)');
  })();
  document.addEventListener('visibilitychange',()=>{if(document.visibilityState==='hidden')flush()});
  setInterval(()=>{if(document.visibilityState==='visible'&&!openSlot)renderSlots()},60000);
})();
</script>
